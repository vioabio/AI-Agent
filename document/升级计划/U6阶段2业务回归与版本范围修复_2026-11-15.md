.\mvnw.cmd -o -B -ntp '-Dvio.build.directory=artifacts/新的整体测试构建' '-Dvio.jaeger.integration=true' '-Dvio.tracing.probe=true' '-Dvio.tracing.probe.output=artifacts/新的SDK探测报告.json' test
